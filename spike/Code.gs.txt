/**
 * FIRS SPIKE TEST - Code.gs  (isolated test backend, NO authentication)
 *
 * Purpose: prove GitHub Pages <-> Apps Script <-> Google Sheets communication.
 * - Container-bound script: create it from Extensions > Apps Script INSIDE the
 *   dedicated test spreadsheet. The spreadsheet ID is never written in code.
 * - Dummy fictional data only. Do NOT point this at the production FIRS sheet.
 *
 * Endpoints (Web App, Execute as: Me, Access: Anyone):
 *   GET  ?action=ping                  -> connectivity + sheet access check
 *   GET  ?action=master                -> dummy master data
 *   GET  ?action=lookup&reportId=...   -> read back a stored report
 *   POST (text/plain, JSON string)     -> {"action":"submitReport", ...}
 */

const CONFIG = {
  VERSION: 'spike-1.0',
  ENABLED: true,            // kill switch: set false and redeploy to stop everything
  MAX_REPORTS: 500,         // hard cap so a public endpoint cannot flood the sheet
  MAX_BODY_CHARS: 50000,
  MAX_ITEMS: 5,
  MAX_TEXT: 200,
  ID_PATTERN: /^SPIKE-[A-Z0-9-]{6,40}$/,
  SHEET_HEADER: 'SPIKE_HEADER',
  SHEET_DETAIL: 'SPIKE_DETAIL'
};

const HEADER_COLUMNS = ['ReportID', 'Timestamp', 'Reporter', 'Area', 'ItemCount', 'AuthMode'];
const DETAIL_COLUMNS = ['ReportID', 'ItemNo', 'Tag', 'Type', 'Part', 'Issue',
                        'ReportType', 'Status', 'OldSerial', 'NewSerial', 'Remark'];

// Fictional data only.
const DUMMY_MASTER = {
  employees: ['Dummy Employee A', 'Dummy Employee B', 'Dummy Employee C'],
  areas: ['Test Area 1', 'Test Area 2'],
  equipment: [
    { tag: 'TEST-CV-0001', type: 'Test Control Valve',  description: 'Dummy control valve 1', area: 'Test Area 1' },
    { tag: 'TEST-CV-0002', type: 'Test Control Valve',  description: 'Dummy control valve 2', area: 'Test Area 1' },
    { tag: 'TEST-PT-0001', type: 'Test Transmitter',    description: 'Dummy transmitter 1',   area: 'Test Area 1' },
    { tag: 'TEST-FT-0001', type: 'Test Flowmeter',      description: 'Dummy flowmeter 1',     area: 'Test Area 2' },
    { tag: 'TEST-PT-0002', type: 'Test Transmitter',    description: 'Dummy transmitter 2',   area: 'Test Area 2' }
  ],
  issues: [
    { type: 'Test Control Valve', part: 'Actuator',    issue: 'Dummy air leakage' },
    { type: 'Test Control Valve', part: 'Positioner',  issue: 'Dummy calibration required' },
    { type: 'Test Transmitter',   part: 'Transmitter', issue: 'Dummy unstable reading' },
    { type: 'Test Transmitter',   part: 'Transmitter', issue: 'Dummy no signal' },
    { type: 'Test Flowmeter',     part: 'Sensor',      issue: 'Dummy no signal' }
  ]
};

/* ------------------------------------------------------------------ */
/* Entry points                                                        */
/* ------------------------------------------------------------------ */

function doGet(e) {
  try {
    if (!CONFIG.ENABLED) return fail_('DISABLED', 'Service is disabled.');

    const params = (e && e.parameter) || {};
    const action = String(params.action || '');

    switch (action) {
      case 'ping':   return ok_(ping_());
      case 'master': return ok_({ master: DUMMY_MASTER });
      case 'lookup': return ok_(lookup_(String(params.reportId || '')));
      default:
        return fail_('UNKNOWN_ACTION', 'Use ?action=ping, ?action=master or ?action=lookup&reportId=...');
    }
  } catch (err) {
    return handleError_(err);
  }
}

function doPost(e) {
  try {
    if (!CONFIG.ENABLED) return fail_('DISABLED', 'Service is disabled.');

    if (!e || !e.postData || typeof e.postData.contents !== 'string') {
      return fail_('BAD_REQUEST', 'Empty request body.');
    }
    if (e.postData.contents.length > CONFIG.MAX_BODY_CHARS) {
      return fail_('BAD_REQUEST', 'Request body too large.');
    }

    let payload;
    try {
      payload = JSON.parse(e.postData.contents);   // body is a JSON string sent as text/plain
    } catch (parseErr) {
      return fail_('BAD_REQUEST', 'Body is not valid JSON.');
    }

    if (!payload || payload.action !== 'submitReport') {
      return fail_('UNKNOWN_ACTION', 'POST action must be "submitReport".');
    }

    // Future: replace with real Google ID token verification. Keep the call site.
    const identity = authorize_(payload);

    return ok_(submitReport_(payload, identity));
  } catch (err) {
    return handleError_(err);
  }
}

/** Maps thrown values to JSON error responses (used by both doGet and doPost). */
function handleError_(err) {
  if (err && err.isValidation) return fail_('VALIDATION', err.message);
  if (err && err.code) return fail_(err.code, err.message);
  return serverError_(err);
}

/* Actions                                                             */

function ping_() {
  const s = sheets_();
  return {
    message: 'pong',
    version: CONFIG.VERSION,
    spreadsheetName: s.ss.getName(),
    storedReports: reportIds_(s.header).length
  };
}

function lookup_(reportId) {
  if (!CONFIG.ID_PATTERN.test(reportId)) {
    throw validation_('Invalid reportId.');
  }

  const s = sheets_();
  const ids = reportIds_(s.header);
  const idx = ids.indexOf(reportId);

  if (idx === -1) return { found: false, reportId: reportId };

  const h = s.header.getRange(idx + 2, 1, 1, HEADER_COLUMNS.length).getValues()[0];

  const lastDetail = s.detail.getLastRow();
  const details = lastDetail < 2 ? [] :
    s.detail.getRange(2, 1, lastDetail - 1, DETAIL_COLUMNS.length).getValues()
      .filter(r => String(r[0]) === reportId)
      .map(r => ({
        itemNo: r[1], tag: r[2], type: r[3], part: r[4], issue: r[5],
        reportType: r[6], status: r[7], oldSerial: r[8], newSerial: r[9], remark: r[10]
      }));

  return {
    found: true,
    reportId: reportId,
    header: {
      timestamp: formatTime_(h[1]),
      reporter: h[2],
      area: h[3],
      itemCount: h[4],
      authMode: h[5]
    },
    details: details
  };
}

function submitReport_(payload, identity) {
  const items = validatePayload_(payload);          // throws on invalid input

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    throw { code: 'BUSY', message: 'System busy. Please retry.' };
  }

  try {
    const s = sheets_();
    const ids = reportIds_(s.header);

    // Idempotency: the same reportId is never written twice.
    if (ids.indexOf(payload.reportId) !== -1) {
      return { reportId: payload.reportId, duplicate: true, itemCount: items.length };
    }

    if (ids.length >= CONFIG.MAX_REPORTS) {
      throw { code: 'CAP_REACHED', message: 'Test storage cap reached. Clear the test sheets.' };
    }

    const now = new Date();
    const rows = items.map((it, i) => [
      payload.reportId, i + 1, it.tag, it.type, it.part, it.issue,
      it.reportType, it.status, it.oldSerial, it.newSerial, it.remark
    ]);

    // Details first, header last (header = commit record).
    // Plain-text format stops formula injection and keeps leading zeros.
    const range = s.detail.getRange(s.detail.getLastRow() + 1, 1, rows.length, DETAIL_COLUMNS.length);
    range.setNumberFormat('@');
    range.setValues(rows);

    s.header.appendRow([
      payload.reportId, now, str_(payload.reporter), str_(payload.area), items.length, identity.mode
    ]);

    SpreadsheetApp.flush();

    return {
      reportId: payload.reportId,
      duplicate: false,
      itemCount: items.length,
      savedAt: formatTime_(now)
    };
  } finally {
    lock.releaseLock();
  }
}

/* ------------------------------------------------------------------ */
/* Validation                                                          */
/* ------------------------------------------------------------------ */

function validatePayload_(p) {
  if (!p || typeof p !== 'object') throw validation_('Payload must be an object.');
  if (!CONFIG.ID_PATTERN.test(str_(p.reportId))) throw validation_('Invalid reportId.');
  if (DUMMY_MASTER.employees.indexOf(str_(p.reporter)) === -1) throw validation_('Invalid reporter.');
  if (DUMMY_MASTER.areas.indexOf(str_(p.area)) === -1) throw validation_('Invalid area.');

  if (!Array.isArray(p.equipment) || p.equipment.length < 1 || p.equipment.length > CONFIG.MAX_ITEMS) {
    throw validation_('Equipment must contain 1 to ' + CONFIG.MAX_ITEMS + ' entries.');
  }

  const reportTypes = ['Finding', 'Replacement', 'Both'];

  return p.equipment.map((item, i) => {
    const n = i + 1;
    if (!item || typeof item !== 'object') throw validation_('Equipment ' + n + ': invalid data.');

    const tag = str_(item.tag);
    const part = str_(item.part);
    const issue = str_(item.issue);
    const reportType = str_(item.reportType);

    const eq = DUMMY_MASTER.equipment.filter(x => x.tag === tag && x.area === str_(p.area))[0];
    if (!eq) throw validation_('Equipment ' + n + ': invalid tag for the selected area.');

    const issueOk = DUMMY_MASTER.issues.some(x => x.type === eq.type && x.part === part && x.issue === issue);
    if (!issueOk) throw validation_('Equipment ' + n + ': invalid part or issue.');

    if (reportTypes.indexOf(reportType) === -1) throw validation_('Equipment ' + n + ': invalid report type.');

    const replacement = reportType === 'Replacement' || reportType === 'Both';
    const status = str_(item.replacementStatus);
    if (replacement && ['Completed', 'Pending'].indexOf(status) === -1) {
      throw validation_('Equipment ' + n + ': invalid replacement status.');
    }

    return {
      tag: eq.tag,
      type: eq.type,
      part: part,
      issue: issue,
      reportType: reportType,
      status: replacement ? status : 'N/A',
      oldSerial: replacement ? str_(item.oldSerial).slice(0, CONFIG.MAX_TEXT) : '',
      newSerial: replacement ? str_(item.newSerial).slice(0, CONFIG.MAX_TEXT) : '',
      remark: str_(item.remark).slice(0, CONFIG.MAX_TEXT)
    };
  });
}

/** Optional: run once from the editor to create the test sheets and authorize the script. */
function setup() {
  const s = sheets_();
  Logger.log('Ready. Spreadsheet: ' + s.ss.getName() + ' | Sheets: ' +
             CONFIG.SHEET_HEADER + ', ' + CONFIG.SHEET_DETAIL);
}

function sheets_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) {
    throw new Error('Script is not bound to a spreadsheet. Create it via Extensions > Apps Script inside the TEST spreadsheet.');
  }
  return {
    ss: ss,
    header: ensureSheet_(ss, CONFIG.SHEET_HEADER, HEADER_COLUMNS),
    detail: ensureSheet_(ss, CONFIG.SHEET_DETAIL, DETAIL_COLUMNS)
  };
}

function ensureSheet_(ss, name, columns) {
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(columns);
    sh.setFrozenRows(1);
  }
  return sh;
}

function reportIds_(headerSheet) {
  const last = headerSheet.getLastRow();
  if (last < 2) return [];
  return headerSheet.getRange(2, 1, last - 1, 1).getValues().map(r => String(r[0]));
}

/* ------------------------------------------------------------------ */
/* Response helpers                                                    */
/* ------------------------------------------------------------------ */

function str_(v) {
  return String(v === null || v === undefined ? '' : v).trim();
}

function validation_(message) {
  return { isValidation: true, message: message };
}

function formatTime_(value) {
  if (value instanceof Date) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ssXXX");
  }
  return String(value);
}

function meta_() {
  return { authMode: 'none', version: CONFIG.VERSION, serverTime: formatTime_(new Date()) };
}

function respond_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function ok_(data) {
  return respond_({ ok: true, data: data, meta: meta_() });
}

function fail_(code, message) {
  return respond_({ ok: false, error: { code: code, message: message }, meta: meta_() });
}

function serverError_(err) {
  console.error(err && err.stack ? err.stack : err);
  // Spike only: message is returned to help debugging. Return a generic text in production.
  return fail_('SERVER_ERROR', err && err.message ? err.message : String(err));
}
